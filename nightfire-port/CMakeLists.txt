if(NIGHTFIRE_CONSUMER135)
    # Keep the pinned checkout untouched. The generated local APU source retains
    # its original LGPL notice; one read-only call is inserted per APU branch.
    set(apu135_original "${XBOXRECOMP_DIR}/src/apu/apu_core.c")
    file(SHA256 "${apu135_original}" apu135_hash)
    if(NOT apu135_hash STREQUAL "0b07b4cc6c8303e314e387315e4d617a6a43b8e6bebfe54271bb25bfe3ca6dd5")
        message(FATAL_ERROR "Consumer135 APU observer requires the verified pinned apu_core.c")
    endif()
    file(READ "${apu135_original}" apu135_text)
    set(apu135_anchor "        if (apu_active && !g_test_tone.active) {")
    string(FIND "${apu135_text}" "${apu135_anchor}" apu135_found)
    if(apu135_found LESS 0)
        message(FATAL_ERROR "Consumer135 APU branch anchor absent")
    endif()
    string(REPLACE "${apu135_anchor}" "${apu135_anchor}\n            nf_consumer135_apu(1, qatomic_read(&d->regs[NV_PAPU_SECTL]), fectl, qatomic_read(&d->regs[NV_PAPU_VPVADDR]), qatomic_read(&d->regs[NV_PAPU_VPSGEADDR]), qatomic_read(&d->regs[NV_PAPU_VPSSLADDR]));" apu135_text "${apu135_text}")
    set(apu135_light "            /* Lightweight: just monitor frame (test tone + software mixer) */")
    string(REPLACE "${apu135_light}" "            nf_consumer135_apu(0, qatomic_read(&d->regs[NV_PAPU_SECTL]), fectl, qatomic_read(&d->regs[NV_PAPU_VPVADDR]), qatomic_read(&d->regs[NV_PAPU_VPSGEADDR]), qatomic_read(&d->regs[NV_PAPU_VPSSLADDR]));\n${apu135_light}" apu135_text "${apu135_text}")
    file(WRITE "${CMAKE_BINARY_DIR}/nightfire_apu_core135.c" "#include \"nightfire_consumer135.h\"\n${apu135_text}")
    get_target_property(apu135_sources xbox_apu SOURCES)
    list(FILTER apu135_sources EXCLUDE REGEX "(^|/)apu_core\\.c$")
    set_property(TARGET xbox_apu PROPERTY SOURCES "${apu135_sources}")
    target_sources(xbox_apu PRIVATE "${CMAKE_BINARY_DIR}/nightfire_apu_core135.c")
    target_include_directories(xbox_apu PRIVATE "${XBOXRECOMP_DIR}/src/apu" "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
    target_compile_definitions(xbox_apu PRIVATE NIGHTFIRE_CONSUMER135)
    target_compile_definitions(xbox_kernel PRIVATE NIGHTFIRE_CONSUMER135)
endif()
